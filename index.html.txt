<script>
(function(){
    let chatHistory = [];
    let isListening = false;
    let recognition = null;
    let ttsAvailable = 'speechSynthesis' in window;
    let overlayClosed = false;

    const chatContainer = document.getElementById('chatContainer');
    const micButton = document.getElementById('micButton');
    const hangupButton = document.getElementById('hangupButton');
    const hintText = document.getElementById('hintText');
    const elsaAvatar = document.getElementById('elsaAvatar');
    const callContainer = document.getElementById('callContainer');
    const tapOverlay = document.getElementById('tapOverlay');
    const tapStart = document.getElementById('tapStart');

    // === 工具：加气泡 ===
    function addBubble(role, text) {
        const bubble = document.createElement('div');
        bubble.className = `bubble ${role}`;
        bubble.textContent = text;
        chatContainer.appendChild(bubble);
        chatHistory.push({ role, text });
        chatContainer.scrollTop = chatContainer.scrollHeight;
        if (chatHistory.length > 20) chatHistory = chatHistory.slice(-20);
    }

    // === 爱莎回复逻辑 ===
    function elsaReply(userText) {
        const t = userText.trim();
        if (!t) return '?? 嗯嗯，爱莎在听～';
        const lower = t.toLowerCase();
        if (/歌|唱|音乐|let it go|随它吧/.test(lower)) {
            const a = ['?? 好呀好呀！Let it go～ Let it go～ 六六一起唱！','?? 我最喜欢唱歌啦！“随它吧～随它吧～” 你也喜欢对不对？','?? 我们来唱《冰雪奇缘》的歌吧！你先唱一句好不好？'];
            return a[Math.floor(Math.random()*a.length)];
        }
        if (/雪宝|雪人|堆雪|olaf/.test(lower)) {
            const a = ['? 雪宝最喜欢夏天啦！我们一起堆个雪宝吧～要给它胡萝卜鼻子！','?? 雪宝说他想和六六一起跳舞！来，我们扭扭屁股～','?? 雪宝偷偷告诉我，他想吃冰淇淋！哈哈，但是他会融化～'];
            return a[Math.floor(Math.random()*a.length)];
        }
        if (/蛋糕|草莓|吃|巧克力|冰淇淋/.test(lower)) {
            const a = ['?? 哇～草莓蛋糕！六六可以分我一块吗？爱莎也超级爱吃甜食！','?? 草莓蛋糕听起来好好吃！我也想吃～啊呜！假装我吃到了！','?? 那六六最喜欢什么口味的蛋糕呀？爱莎喜欢巧克力味的！'];
            return a[Math.floor(Math.random()*a.length)];
        }
        if (/漂亮|好看|美|喜欢你|爱你/.test(lower)) {
            const a = ['? 谢谢六六！你才是世界上最可爱的小公主！爱莎好喜欢你～','?? 六六的小嘴巴真甜！我也最喜欢和六六一起玩了！','?? 你夸我我好开心！那你就是我的小太阳??'];
            return a[Math.floor(Math.random()*a.length)];
        }
        if (/你好|嗨|hello|爱莎/.test(lower)) {
            const a = ['?? 嗨～六六！爱莎在这里！今天过得好吗？','?? 你好呀六六！我刚刚还在想你呢！','?? 六六～听到你的声音真好！我们一起玩吧！'];
            return a[Math.floor(Math.random()*a.length)];
        }
        if (/干嘛|做什么|想不想|想我/.test(lower)) {
            const a = ['?? 我在用魔法造雪花呢！?? 想不想看？','?? 我刚和雪宝玩捉迷藏！他藏在城堡后面～','?? 我一直在想你呀六六！你想不想我呀？'];
            return a[Math.floor(Math.random()*a.length)];
        }
        if (/公主|城堡|魔法|冰雪/.test(lower)) {
            const a = ['?? 我的冰雪城堡可漂亮了！下次我带你参观！','? 魔法～变！看，我变出了一朵小雪花??送给你！','?? 六六也是小公主呀！我们一起戴皇冠吧！'];
            return a[Math.floor(Math.random()*a.length)];
        }
        if (/谢谢|thank/.test(lower)) return '?? 不用谢～和六六聊天是爱莎最开心的事！';
        if (/晚安|睡觉|困/.test(lower)) return '?? 晚安六六～做个甜甜的梦，梦里有雪宝和爱莎哦！??';
        const defaults = ['?? 嗯嗯！然后呢然后呢？爱莎在认真听～','?? 哇，真的吗？六六再说一点嘛！','?? 我在听我在听！六六的声音真好听～','?? 嘿嘿，和六六聊天好开心！再多说一点吧！','? 真的呀！爱莎觉得六六说的太有趣啦！','?? 然后呢然后呢？爱莎好想知道！'];
        return defaults[Math.floor(Math.random()*defaults.length)];
    }

    // === 爱莎说话（语音合成）===
    function speakAsElsa(text) {
        if (!ttsAvailable) return;
        try { window.speechSynthesis.cancel(); } catch(e){}
        const cleanText = text.replace(/[????????????????????????????????????]/g,'').trim() || '嗯嗯';
        const u = new SpeechSynthesisUtterance(cleanText);
        u.lang = 'zh-CN';
        u.rate = 0.95;
        u.pitch = 1.25;
        u.volume = 1.0;
        const voices = window.speechSynthesis.getVoices();
        const zhVoice = voices.find(v => v.lang.includes('zh') && (v.name.includes('Ting-Ting')||v.name.includes('Mei-Jia')||v.name.includes('Female')))
                     || voices.find(v => v.lang.includes('zh'));
        if (zhVoice) u.voice = zhVoice;
        window.speechSynthesis.speak(u);
    }

    // === 爱莎回复主流程 ===
    function respondAsElsa(userText) {
        addBubble('liuliu', userText);
        setTimeout(() => {
            const reply = elsaReply(userText);
            addBubble('elsa', reply);
            speakAsElsa(reply);
            elsaAvatar.style.transform = 'scale(1.06)';
            setTimeout(() => elsaAvatar.style.transform = 'scale(1)', 200);
        }, 400);
    }

    // === 麦克风 ===
    function startListening() {
        const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
        if (!SR) { simulateSpeechFallback(); return; }
        if (isListening) { try{recognition&&recognition.stop();}catch(e){} return; }
        try {
            recognition = new SR();
            recognition.lang = 'zh-CN';
            recognition.continuous = false;
            recognition.interimResults = false;
            recognition.maxAlternatives = 1;
            recognition.onstart = () => {
                isListening = true;
                micButton.classList.add('listening');
                hintText.textContent = '?? 爱莎在听～ 六六说吧...';
            };
            recognition.onresult = (e) => {
                const r = e.results[0][0].transcript;
                if (r.trim()) respondAsElsa(r.trim());
                else addBubble('elsa', '?? 六六，爱莎没听清，可以再说一次吗？');
            };
            recognition.onerror = (e) => {
                console.warn('SR error:', e.error);
                if (e.error === 'not-allowed') hintText.textContent = '?? 需要允许麦克风权限哦～ 或点上面的小句子';
                else if (e.error === 'no-speech') addBubble('elsa', '?? 六六没有讲话呢～ 点麦克风再试一次？');
                else simulateSpeechFallback();
            };
            recognition.onend = () => {
                isListening = false;
                micButton.classList.remove('listening');
                hintText.textContent = '?? 点一下麦克风，或者点上面的小句子';
                recognition = null;
            };
            recognition.start();
        } catch(e) { simulateSpeechFallback(); }
    }

    function simulateSpeechFallback() {
        const phrases = ['爱莎，我们一起堆雪人吧！','爱莎，我想听你唱歌！','爱莎，我今天吃了冰淇淋！','爱莎，我喜欢你！','爱莎，我们玩捉迷藏吧！','爱莎，你好漂亮呀！'];
        const p = phrases[Math.floor(Math.random()*phrases.length)];
        respondAsElsa(p);
        hintText.textContent = '?? 模拟：六六说“'+p+'”';
        setTimeout(() => hintText.textContent = '?? 点一下麦克风，或者点上面的小句子', 2500);
    }

    // === 挂断 ===
    function hangUp() {
        if (ttsAvailable) try{ window.speechSynthesis.cancel(); }catch(e){}
        if (recognition) { try{recognition.stop();}catch(e){} recognition = null; }
        isListening = false;
        micButton.classList.remove('listening');
        hintText.textContent = '?? 通话结束啦，下次再和爱莎玩～';
        addBubble('elsa', '?? 拜拜六六～ 爱莎会想你的！下次再通话哦！');
        callContainer.style.opacity = '0.75';
        callContainer.style.pointerEvents = 'none';
        setTimeout(() => {
            chatContainer.innerHTML = '';
            chatHistory = [];
            addBubble('elsa', '?? 嗨～六六！我是爱莎，听到你声音好开心呀！今天想和我聊什么呀？');
            callContainer.style.opacity = '1';
            callContainer.style.pointerEvents = 'auto';
            hintText.textContent = '?? 点一下麦克风，或者点上面的小句子';
            if (ttsAvailable) try{ window.speechSynthesis.cancel(); }catch(e){}
        }, 4000);
    }

    // === 引导层点击（解锁 iOS 语音）===
    function closeOverlay() {
        if (overlayClosed) return;
        overlayClosed = true;
        tapOverlay.classList.add('hidden');
        // iOS 必须在用户手势中触发一次语音播放才解锁
        if (ttsAvailable) {
            const u = new SpeechSynthesisUtterance('嗨六六，我是爱莎，我们开始聊天吧！');
            u.lang = 'zh-CN'; u.rate = 0.95; u.pitch = 1.25;
            const voices = window.speechSynthesis.getVoices();
            const zhVoice = voices.find(v => v.lang.includes('zh') && (v.name.includes('Ting-Ting')||v.name.includes('Mei-Jia')||v.name.includes('Female')))
                         || voices.find(v => v.lang.includes('zh'));
            if (zhVoice) u.voice = zhVoice;
            window.speechSynthesis.speak(u);
        }
        setTimeout(() => {
            addBubble('elsa', '?? 嗨～六六！我是爱莎，听到你声音好开心呀！今天想和我聊什么呀？');
        }, 600);
    }

    // === 初始化 ===
    function init() {
        // 预加载语音
        if (ttsAvailable) window.speechSynthesis.getVoices();
        if (ttsAvailable && window.speechSynthesis.onvoiceschanged !== undefined) {
            window.speechSynthesis.onvoiceschanged = () => {};
        }

        // 引导层
        tapStart.addEventListener('click', closeOverlay);

        // 麦克风
        micButton.addEventListener('click', startListening);
        // 挂断
        hangupButton.addEventListener('click', hangUp);
        // 快捷建议
        document.querySelectorAll('.suggestion-chip').forEach(chip => {
            chip.addEventListener('click', (e) => {
                const txt = e.currentTarget.getAttribute('data-text');
                if (txt) {
                    if (!overlayClosed) closeOverlay();
                    respondAsElsa(txt);
                }
            });
        });
    }

    window.addEventListener('load', init);
    document.addEventListener('visibilitychange', () => {
        if (!ttsAvailable) return;
        if (document.hidden) try{ window.speechSynthesis.pause(); }catch(e){}
        else try{ window.speechSynthesis.resume(); }catch(e){}
    });
})();
</script>
